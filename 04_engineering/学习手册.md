"""所有模型调用的唯一出口:聊天走 DeepSeek,向量走智谱"""
import os

from dotenv import load_dotenv
from openai import OpenAI

load_dotenv()

chat_client = OpenAI(
    api_key=os.environ.get("OPENAI_API_KEY"),
    base_url=os.environ.get("OPENAI_BASE_URL"),
)
CHAT_MODEL = os.environ.get("MODEL_NAME")

embed_client = OpenAI(
    api_key=os.environ.get("EMBEDDING_API_KEY"),
    base_url=os.environ.get("EMBEDDING_BASE_URL"),
)
EMBEDDING_MODEL = os.environ.get("EMBEDDING_MODEL", "embedding-3")


def chat(messages, tools=None, **kwargs):
    """非流式聊天,返回 SDK 响应对象;kwargs 原样透传(比如 response_format)"""
    return chat_client.chat.completions.create(
        model=CHAT_MODEL, messages=messages, tools=tools, **kwargs
    )


def embed_texts(texts, batch_size=32):
    """批量向量化,一次最多 32 条"""
    vectors = []
    for start in range(0, len(texts), batch_size):
        batch = texts[start:start + batch_size]
        resp = embed_client.embeddings.create(model=EMBEDDING_MODEL, input=batch)
        vectors.extend(item.embedding for item in resp.data)
    return vectors
```

"""Agent 循环:点菜 → 执行 → 喂回,直到给出最终回答。不藏任何全局状态"""
import json

import llm_client
from tools import TOOLS, TOOL_IMPLS

MAX_STEPS = 10


def run_tool_calls(assistant_msg, history):
    for tool_call in assistant_msg.tool_calls:
        name = tool_call.function.name
        try:
            args = json.loads(tool_call.function.arguments)
        except json.JSONDecodeError:
            args = {}
        func = TOOL_IMPLS.get(name)
        if func is None:
            result = f"错误:不存在名为 {name} 的工具"
        else:
            try:
                result = func(**args)
            except Exception as e:
                result = f"工具执行出错:{e}"
        print(f"[工具] {name}({args}) -> {result}")
        history.append({"role": "tool", "tool_call_id": tool_call.id, "content": str(result)})


def reply(user_input, history):
    """处理一句话:循环点菜直到最终回答。history 由调用方持有并原地追加"""
    history.append({"role": "user", "content": user_input})
    for _ in range(MAX_STEPS):
        response = llm_client.chat(history, tools=TOOLS)
        msg = response.choices[0].message
        history.append(msg.model_dump(exclude_none=True))
        if not msg.tool_calls:
            return msg.content
        run_tool_calls(msg, history)
    return "(达到最大步数,强制停止)"
```

## 第 2 课 · @tool 装饰器:注册表自动化(本阶段核心)

**目标**:用装饰器 + `inspect` 反射,把"函数 → JSON Schema → 注册表"三件事压缩成一个 `@tool`。加新工具 = 写一个带 `@tool` 的函数,**其他任何文件一行不改**。你练过的 `00_python基础/05_装饰器.py` 在这里派上用场。

**新概念**:
- **装饰器 = Java 注解 + 反射扫描的合体**:`@tool` 是注解,`inspect.signature` 是反射,Spring 启动时扫 Bean 干的是同一件事
- **docstring 就是 description**:模型看不见你的代码,只看得见说明书——写在函数下方三引号里,位置对了它就自动化
- **开闭原则**:对扩展开放(加函数),对修改关闭(注册器、agent、main 全都不动)

**`tools.py` 整体重写**:

```python
"""工具注册器:@tool 一挂,函数自动变成'模型可点的菜'"""
import inspect
from datetime import datetime

# 注册表:name -> {"func": 函数, "doc": 说明书, "params": JSON Schema}
TOOL_REGISTRY = {}

_TYPE_MAP = {str: "string", int: "integer", float: "number", bool: "boolean"}


def _schema_from_signature(func):
    """反射函数签名 → JSON Schema。有类型注解用注解,没有默认 string"""
    properties, required = {}, []
    for name, param in inspect.signature(func).parameters.items():
        ann = param.annotation if param.annotation is not inspect.Parameter.empty else str
        properties[name] = {"type": _TYPE_MAP.get(ann, "string")}
        if param.default is inspect.Parameter.empty:
            required.append(name)
    return {"type": "object", "properties": properties, "required": required}


def tool(func):
    """@tool:登记函数、自动生成 Schema。原样返回 func,函数行为不变"""
    TOOL_REGISTRY[func.__name__] = {
        "func": func,
        "doc": inspect.getdoc(func) or "",
        "params": _schema_from_signature(func),
    }
    return func


def get_openai_tools():
    """给 API 的 tools 参数:从注册表拼出协议格式"""
    return [
        {"type": "function",
         "function": {"name": name, "description": meta["doc"], "parameters": meta["params"]}}
        for name, meta in TOOL_REGISTRY.items()
    ]


def call_tool(name, args: dict):
    """统一执行出口:找不到工具、参数解析错、执行报错,全部兜底成字符串"""
    meta = TOOL_REGISTRY.get(name)
    if meta is None:
        return f"错误:不存在名为 {name} 的工具"
    try:
        return str(meta["func"](**args))
    except Exception as e:
        return f"工具执行出错:{e}"


# ================= 工具定义区:加工具只改这里 =================

@tool
def get_current_time() -> str:
    """获取当前的日期和时间"""
    return datetime.now().strftime("%Y-%m-%d %H:%M:%S %A")


@tool
def write_file(path: str, content: str) -> str:
    """把文本内容写入指定文件(覆盖原内容)"""
    with open(path, "w", encoding="utf-8") as f:
        f.write(content)
    return f"已写入 {path},共 {len(content)} 个字符"


@tool
def read_file(path: str) -> str:
    """读取指定文件的全部内容"""
    with open(path, "r", encoding="utf-8") as f:
        return f.read()
```

```python
def chunk_text(text, max_len=500):
    """按空行分段(一段一个独立事实),超长段落再用滑窗细切"""
    chunks = []
    for para in text.split("\n\n"):
        para = para.strip()
        if not para:
            continue
        if len(para) <= max_len:
            chunks.append(para)
        else:
            chunks.extend(chunk_by_size(para, size=400, overlap=50))
    return chunks
```

## 第 4 课 · Pydantic 结构化输出:模型输出也该有"DTO"

**目标**:让模型返回 Pydantic 对象——字段、类型自动校验,**不合格自动打回重写**。顺手解决一个老问题:用结构化输出做意图路由。

**新概念**:
- `BaseModel` = Java 的 DTO + Bean Validation 注解合体;`Field(description=...)` 校验之外还自动生成给模型的说明书
- `model_cls.model_json_schema()` → 直接塞进 system prompt:**入参用 JSON Schema(tools),出参也用 JSON Schema(pydantic),一门协议两头用**
- **校验失败不崩程序,把错误喂回模型重写**——和阶段 2"工具报错喂回去"同一个思想
- Java:`Jackson 反序列化 + @NotNull/@Pattern`,只不过这里的"校验器失败"能跟模型对话

**`structured.py`**:

```python
"""结构化输出:模型返回 Pydantic 对象,校验不过自动重试"""
from pydantic import BaseModel, Field, ValidationError

import llm_client


class Reply(BaseModel):
    """演示用:带意图识别的回答"""
    answer: str = Field(description="对用户问题的简短回答")
    intent: str = Field(description="意图:kb(查公司知识库)/time(时间)/file(文件操作)/chat(闲聊)之一")
    need_search: bool = Field(description="是否需要查询知识库")


def ask_structured(user_text, model_cls, retries=2):
    """带自动重试的结构化调用。返回 (对象 or None, 实际尝试次数)"""
    system = (
        "你是结构化输出器。只输出一个 JSON 对象,不要输出任何其他文字。"
        "字段定义(JSON Schema):\n"
        f"{model_cls.model_json_schema()}"
    )
    messages = [
        {"role": "system", "content": system},
        {"role": "user", "content": user_text},
    ]
    for attempt in range(retries + 1):
        resp = llm_client.chat(messages, response_format={"type": "json_object"})
        raw = resp.choices[0].message.content
        try:
            return model_cls.model_validate_json(raw), attempt
        except ValidationError as e:
            bad = e.errors()[0]
            print(f"(第 {attempt + 1} 次不合格:{bad['loc']} {bad['msg']},打回重写)")
            messages.append({"role": "assistant", "content": raw})
            messages.append({"role": "user",
                             "content": "上面的输出不符合 JSON Schema,修正后重新只输出 JSON。"})
    return None, retries


if __name__ == "__main__":
    while True:
        text = input("\n问一句(直接回车退出): ").strip()
        if not text:
            break
        obj, attempt = ask_structured(text, Reply)
        if obj is None:
            print("模型始终不合格,放弃")
            continue
        print(f"回答:{obj.answer}\n意图:{obj.intent} | 需要查库:{obj.need_search} | 尝试 {attempt + 1} 次")
```

> 若报错提示 `response_format` 不支持:删掉那个参数,只靠 prompt 约束(阶段 1 第 6 课的老结论:提示词约束不可靠,所以才需要校验 + 重试兜底)。

## 第 5 课 · ReAct 模式:不用协议也能跑 Agent

**目标**:写一个**不使用 function calling 协议**的 Agent——模型靠提示词输出 `Thought / Action / Observation` 文本,你的代码用字符串解析驱动循环。跑通后你会彻底理解:Agent 的本质是循环,协议只是让循环更可靠。

**新概念**:
- **ReAct**(Reason + Act,2022):Agent 思想原型。function calling 就是从它进化来的协议化版本
- **Observation 以 role=user 喂回**——任何能聊天的模型都能跑,这是它唯一的优点
- **解析是软肋**:模型多写一个字格式就崩,所以要"格式纠正消息"兜底
- 结论先行:**生产用 function calling,ReAct 用来理解原理 + 兼容没有工具协议的模型**

**`react_agent.py`**(单文件对比课,直接复用 tools.py 的注册器——第 2 课的回报):

```python
"""ReAct:纯提示词驱动的 Agent。对比 function calling 理解协议的价值"""
import json

import llm_client
from tools import TOOL_REGISTRY, call_tool

MAX_STEPS = 6


def build_tool_menu():
    lines = []
    for name, meta in TOOL_REGISTRY.items():
        params = ", ".join(meta["params"]["properties"].keys())
        lines.append(f"- {name}({params}):{meta['doc']}")
    return "\n".join(lines)


def react_reply(user_input, history):
    history.append({"role": "user", "content": user_input})
    system = f"""你是使用工具的助手。严格按以下两种格式之一回答,不要输出别的内容:

Thought: <一句话思考该做什么>
Action: <工具名,必须是下面菜单之一>
Action Input: <JSON 对象,是工具参数,没有参数写 {{}}>

或(已经有答案时):

Thought: <一句话思考>
Final Answer: <给用户的最终回答>

可用工具菜单:
{build_tool_menu()}"""
    history.insert(0, {"role": "system", "content": system})

    for step in range(MAX_STEPS):
        text = llm_client.chat(history).choices[0].message.content
        history.append({"role": "assistant", "content": text})
        print(f"--- 第 {step + 1} 步 ---\n{text}")

        if "Final Answer:" in text:
            return text.split("Final Answer:", 1)[1].strip()

        action, action_input = parse_action(text)
        if action is None:
            history.append({"role": "user",
                            "content": "格式不对!必须严格按 Thought/Action/Action Input 或 Final Answer 输出。"})
            continue

        result = call_tool(action, action_input)
        print(f"[工具] {action}({action_input}) -> {result}")
        history.append({"role": "user", "content": f"Observation: {result}"})
    return "(达到最大步数,强制停止)"


def parse_action(text):
    """从模型文本里抠出 Action 和 Action Input——ReAct 的脆弱之处就在这"""
    action, action_input = None, {}
    for line in text.splitlines():
        line = line.strip()
        if line.startswith("Action:") and action is None:
            action = line.split("Action:", 1)[1].strip()
        elif line.startswith("Action Input:"):
            try:
                action_input = json.loads(line.split("Action Input:", 1)[1].strip())
            except json.JSONDecodeError:
                action_input = {}
    return action, action_input


if __name__ == "__main__":
    history = []
    while True:
        text = input("\n你: ").strip()
        if not text:
            break
        print(f"AI: {react_reply(text, history)}")
```

1. 问"现在几点,再把结果写进 time.txt"——数它走了几步,每步 Observation 是怎么喂回去的
2. 问"用掷骰子工具掷一下"——菜单里没有,看它编 Action 还是老实说没有
3. 故意把 `Action Input` 的 JSON 写错(在 prompt 里诱导),体验"格式崩"和纠正消息的作用

## 第 6 课 · 流式 Agent:边想边说

**目标**:`llm_client` 新增 `chat_stream`,Agent 循环改成流式——回答一个字一个字往外蹦,工具调用以 `[工具]` 卡片形式出现。真实聊天产品的形态。

**新概念**:
- 流式下 **tool_calls 是碎片**:名字先到,`arguments` 一段段补——必须按 `index` 攒,攒完才有完整点菜单
- 思考型模型(deepseek-flash)会先流 `reasoning_content`(草稿纸),本课直接忽略
- **流式把消息从"SDK 对象"变成了"你自己攒的字典"**——意外收获:第 7 课 mock 起来简单十倍

**`llm_client.py` 追加**:

```python
def chat_stream(messages, tools=None):
    """流式聊天:边生成边打印;返回攒好的 (content, tool_calls 列表)"""
    stream = chat_client.chat.completions.create(
        model=CHAT_MODEL, messages=messages, tools=tools, stream=True
    )
    content_parts = []
    tc_slots = {}   # index -> {"id","name","arguments"}:按碎片顺序攒

    for chunk in stream:
        if not chunk.choices:
            continue
        delta = chunk.choices[0].delta
        if delta.content:
            print(delta.content, end="", flush=True)
            content_parts.append(delta.content)
        for tc in (delta.tool_calls or []):
            slot = tc_slots.setdefault(tc.index, {"id": "", "name": "", "arguments": ""})
            if tc.id:
                slot["id"] = tc.id
            if tc.function and tc.function.name:
                slot["name"] = tc.function.name
            if tc.function and tc.function.arguments:
                slot["arguments"] += tc.function.arguments

    print()
    tool_calls = [
        {"id": s["id"], "type": "function",
         "function": {"name": s["name"], "arguments": s["arguments"]}}
        for _, s in sorted(tc_slots.items())
    ]
    return "".join(content_parts), tool_calls
```

**`agent.py` 最终版**(整个文件替换;从此消息是自己攒的字典,不再依赖 SDK 对象):

```python
"""Agent 循环:点菜 → 执行 → 喂回,直到给出最终回答"""
import json

import llm_client
from tools import call_tool, get_openai_tools

MAX_STEPS = 10


def run_tool_calls(tool_calls, history):
    """执行模型点的所有菜(现在是纯字典,不再依赖 SDK 对象)"""
    for tc in tool_calls:
        name = tc["function"]["name"]
        try:
            args = json.loads(tc["function"]["arguments"])
        except json.JSONDecodeError:
            args = {}
        result = call_tool(name, args)
        print(f"[工具] {name}({args}) -> {result}")
        history.append({"role": "tool", "tool_call_id": tc["id"], "content": str(result)})


def reply(user_input, history):
    """处理一句话:流式循环点菜直到最终回答。history 由调用方持有"""
    history.append({"role": "user", "content": user_input})

    for _ in range(MAX_STEPS):
        content, tool_calls = llm_client.chat_stream(history, tools=get_openai_tools())

        if tool_calls:
            history.append({"role": "assistant", "content": content or None,
                            "tool_calls": tool_calls})
            run_tool_calls(tool_calls, history)
        else:
            history.append({"role": "assistant", "content": content})
            return content

    return "(达到最大步数,强制停止)"
```

## 第 7 课 · pytest:0 token 的回归测试

**目标**:给注册器、Agent 循环、结构化重试写单测——**mock 掉 LLM,一分钱不花**,改坏任何东西立刻知道。这是"工程化"的最后一块,也是你和"脚本人"的分水岭。

**新概念**:
- **单测测的是你的逻辑,不是模型的智能**。LLM 是外部依赖,和数据库一样要 mock(Java:Mockito)
- **mock 的位置 = 依赖的边界**:agent 依赖 `llm_client.chat_stream`,那就 patch 这个函数,给它一份"固定剧本"
- `monkeypatch.setattr(模块, "名字", 假货)` 是 pytest 的注入器;`tmp_path` 是内置的临时目录 fixture
- 命名约定:文件 `test_*.py`、函数 `test_*`,pytest 靠名字自动发现

## 常见报错速查

| 报错/现象 | 原因 | 解决 |
|---|---|---|
| `ModuleNotFoundError: No module named 'agent'`(跑 pytest 时) | 直接敲 `pytest` 没把项目目录加进 sys.path | 用 `python -m pytest tests -v`(cwd 进 sys.path) |
| `ImportError: cannot import name ... from partially initialized module` | 循环导入(比如 tools import 了 agent) | 检查依赖方向:main → agent → tools/llm_client,禁止反向 |
| `ValidationError`(pydantic) | 模型输出不符合 Schema | 看 `e.errors()[0]` 的 loc/msg;确认 Schema 塞进了 prompt;重试兜底已处理 |
| 装饰器后函数"变了" | `@tool` 里忘了 `return func` | 返回原函数,只要登记副作用 |
| 模型从不点某个工具 | docstring 没写 / description 太泛 | `test_每个工具都有说明书` 已经在帮你把关 |
| 流式 `tool_calls` 拼不完整 | 忘了按 `tc.index` 分槽攒 arguments | 对照第 6 课 `chat_stream` |
| `monkeypatch` 不生效 | patch 错了路径 | patch "被使用的地方"(如 `agent.llm_client.chat_stream`) |
| pytest 说收集到 0 个用例 | 文件/函数名不符合约定 | 文件 `test_*.py`,函数 `test_*` |
| `知识库目录是空的` | `03_rag/knowledge` 被移动或改名 | 恢复目录;或改 rag.py 的 `KNOWLEDGE_DIR` |

**卡住的标准动作不变**:读报错最后一行 → 搜索 → 超 30 分钟,代码 + 完整报错 + 你跑的原话一起贴给 ZCode。
